\section{Descripción del Problema}
%-----------------------------------------------------------------------

El \textbf{Problema de Portfolio} consiste en determinar la asignación óptima de capital
entre $N$ activos financieros con el objetivo de maximizar el retorno esperado
minimizando simultáneamente el riesgo.

\subsection*{Representación de los datos}

Para cada activo $i \in \{1,\ldots,N\}$ y día $t \in \{1,\ldots,T\}$ se dispone
del retorno diario $\mu_{ti}$, calculado como:
\[
  \mu_{ti} = \frac{V_{ti} - V_{(t+1)i}}{V_{ti}}
\]
Los datos cubren el periodo 2015--2024 para entrenamiento y el año 2025 para evaluación.

\subsection*{Función objetivo}

Dado un vector de pesos $w = (w_1,\ldots,w_N)$:
\[
  \text{fitness}(w) = \text{Beneficio}(w) - \lambda \cdot \text{Riesgo}(w)
\]
\[
  \text{Beneficio}(w) = \sum_{i=1}^{N} w_i \cdot \sum_{t=1}^{T}\log(1+\mu_{ti})
  \qquad
  \text{Riesgo}(w) = \sqrt{\sum_{i=1}^{N}\sum_{j=1}^{N} w_i \cdot \Sigma_{ij} \cdot w_j}
\]
siendo $\Sigma_{ij}$ la covarianza entre activos $i$ y $j$, y $\lambda = 500$.

\subsection*{Restricciones}

\[
  \sum_{i=1}^{N} w_i = 1 \qquad
  w_i = 0 \;\vee\; lo \le w_i \le hi \quad \forall i
\]

%-----------------------------------------------------------------------
\section{Aplicación de los Algoritmos al Problema}
%-----------------------------------------------------------------------

\subsection{Representación de soluciones}

Una solución es un vector $w \in \mathbb{R}^N$ donde $w_i$ indica la fracción de capital
invertida en el activo $i$: exactamente cero o en $[lo, hi]$.

\subsection{Generación aleatoria de soluciones}

Igual que en la Práctica 1: subconjunto aleatorio de activos activos, pesos aleatorios en
$[lo, hi]$, normalización y reparación.

\subsection{Operador de reparación (\texttt{fix})}

Tres fases: normalización, recorte de rango y redistribución iterativa. Es esencial tras
los operadores de cruce, que pueden generar soluciones infactibles.

\subsection{Operador de selección: torneo binario}

Se eligen dos individuos al azar y se devuelve el de mayor fitness.

\begin{algorithm}[H]
\caption{Torneo binario}
\KwIn{población $P$} \KwOut{individuo ganador}
$i_1 \leftarrow$ aleatorio en $[0,|P|-1]$;\quad $i_2 \leftarrow$ aleatorio en $[0,|P|-1]$\;
\Si{$P[i_1].\text{fitness} \geq P[i_2].\text{fitness}$}{\KwReturn{$P[i_1]$}}
\KwReturn{$P[i_2]$}
\end{algorithm}

\subsection{Operadores de cruce}

\subsubsection*{Cruce Aritmético}
Para cada posición $i$ se genera $r_i \in [0,1]$ independiente:
\[
  \text{Off}^1_i = r_i \cdot S^1_i + (1-r_i) \cdot S^2_i
  \qquad
  \text{Off}^2_i = r_i \cdot S^2_i + (1-r_i) \cdot S^1_i
\]

\begin{algorithm}[H]
\caption{Cruce Aritmético($S^1$, $S^2$, problema)}
\KwIn{padres $S^1$, $S^2$; $[lo,hi]$} \KwOut{hijos $\text{Off}^1$, $\text{Off}^2$}
$\text{Off}^1 \leftarrow S^1$;\quad $\text{Off}^2 \leftarrow S^2$\;
\Para{$i \leftarrow 1$ \KwTo $N$}{
  $r \leftarrow$ aleatorio en $[0,1]$\;
  $\text{Off}^1[i] \leftarrow r \cdot S^1[i] + (1-r) \cdot S^2[i]$\;
  $\text{Off}^2[i] \leftarrow r \cdot S^2[i] + (1-r) \cdot S^1[i]$\;
}
\fix{$\text{Off}^1$};\quad \fix{$\text{Off}^2$};\quad \KwReturn{$\text{Off}^1$, $\text{Off}^2$}
\end{algorithm}

\subsubsection*{Cruce BLX-$\alpha$}
Con $I_i = |S^1_i - S^2_i|$:
\[
  hmin_i = \min(S^1_i, S^2_i) - \alpha \cdot I_i \qquad
  hmax_i = \max(S^1_i, S^2_i) + \alpha \cdot I_i
\]
Ambos hijos se generan uniformemente en $[\max(0,hmin_i),\min(hi,hmax_i)]$, $\alpha=0{,}3$.

\begin{algorithm}[H]
\caption{Cruce BLX-$\alpha$($S^1$, $S^2$, problema), $\alpha=0{,}3$}
\KwIn{padres $S^1$, $S^2$; $[lo,hi]$; $\alpha$} \KwOut{hijos $\text{Off}^1$, $\text{Off}^2$}
\Para{$i \leftarrow 1$ \KwTo $N$}{
  $I \leftarrow |S^1[i] - S^2[i]|$\;
  $hmin \leftarrow \max(0,\;\min(S^1[i], S^2[i]) - \alpha \cdot I)$\;
  $hmax \leftarrow \min(hi,\;\max(S^1[i], S^2[i]) + \alpha \cdot I)$\;
  $\text{Off}^1[i] \leftarrow$ aleatorio en $[hmin, hmax]$\;
  $\text{Off}^2[i] \leftarrow$ aleatorio en $[hmin, hmax]$\;
}
\fix{$\text{Off}^1$};\quad \fix{$\text{Off}^2$};\quad \KwReturn{$\text{Off}^1$, $\text{Off}^2$}
\end{algorithm}

\subsection{Operador de mutación}

\subsubsection*{Para Algoritmos Genéticos}
Con probabilidad $p_m=0{,}1$ por individuo se selecciona un gen aleatorio y se reemplaza
por un valor uniforme en $[lo,hi]$, seguido de \fix{}. Solo cuenta como evaluación si hay mutación.

\begin{algorithm}[H]
\caption{Mutación AG($ind$, problema)}
\Si{rand() $\leq 0{,}1$}{
  $gen \leftarrow$ aleatorio en $[0, N-1]$\;
  $ind[gen] \leftarrow$ aleatorio en $[lo, hi]$\;
  \fix{$ind$};\quad $evals \leftarrow evals + 1$\;
}
\end{algorithm}

\subsubsection*{Para Algoritmos Meméticos}
Con probabilidad $p_m=0{,}1/N$ por gen se reemplaza independientemente. Solo cuenta como
evaluación si se muta algún gen.

\begin{algorithm}[H]
\caption{Mutación AM($ind$, problema)}
$mutado \leftarrow$ falso\;
\Para{$i \leftarrow 1$ \KwTo $N$}{
  \Si{rand() $\leq 0{,}1/N$}{
    $ind[i] \leftarrow$ aleatorio en $[lo, hi]$;\quad $mutado \leftarrow$ verdadero\;
  }
}
\Si{$mutado$}{
  \fix{$ind$};\quad $evals \leftarrow evals + 1$\;
}
\end{algorithm}

\section{Algoritmos Genéticos y Meméticos}

\subsection{Algoritmo Genético Generacional (AGG)}

\subsubsection*{Parámetros}
$N_p=100$, $p_c=0{,}8$, $p_m=0{,}1$ por individuo, criterio de parada: $10\,000$ evaluaciones.

\subsubsection*{Esquema de evolución y reemplazamiento}

\begin{algorithm}[H]
\caption{AGG (generacional con elitismo)}
\KwIn{problema, maxEvals} \KwOut{mejor solución}
Inicializar $P$ con $N_p$ individuos aleatorios; $evals \leftarrow N_p$\;
\Mientras{$evals < \text{maxEvals}$}{
  $P' \leftarrow \emptyset$\;
  \Para{$i \leftarrow 1$ \KwTo $N_p/2$ \textbf{y} $evals < \text{maxEvals}$}{
    $p_1 \leftarrow$ \torneo{$P$}; $p_2 \leftarrow$ \torneo{$P$}\;
    \Si{rand() $\leq 0{,}8$}{$(h_1,h_2)\leftarrow$\cruzar{$p_1,p_2$}; $evals \mathrel{+}= 2$}
    \SiNo{$h_1\leftarrow p_1$; $h_2\leftarrow p_2$}
    \mutar{$h_1$}; \mutar{$h_2$}; $P'\leftarrow P'\cup\{h_1,h_2\}$\;
  }
  \Si{$\max(P').\text{fitness} < \max(P).\text{fitness}$}{$\min(P')\leftarrow\max(P)$}
  $P \leftarrow P'$\;
}
\KwReturn{$\max(P)$}
\end{algorithm}

Se implementan dos variantes: \textbf{AGG-Arit} y \textbf{AGG-BLX}.

\subsection{Algoritmo Genético Estacionario (AGE)}

\subsubsection*{Parámetros}
$N_p=100$, $p_c=1{,}0$ (siempre se cruzan), $p_m=0{,}1$ por individuo,
criterio de parada: $10\,000$ evaluaciones.

\subsubsection*{Esquema de evolución y reemplazamiento}

\begin{algorithm}[H]
\caption{AGE (estacionario)}
\KwIn{problema, maxEvals} \KwOut{mejor solución}
Inicializar $P$ con $N_p$ individuos aleatorios; $evals \leftarrow N_p$\;
\Mientras{$evals < \text{maxEvals}$}{
  $p_1\leftarrow$\torneo{$P$}; $p_2\leftarrow$\torneo{$P$}\;
  $(h_1,h_2)\leftarrow$\cruzar{$p_1,p_2$}; $evals \mathrel{+}= 2$\;
  \mutar{$h_1$}; \mutar{$h_2$}\;
  \Si{$h_1.\text{fitness}>\min(P).\text{fitness}$}{$\min(P)\leftarrow h_1$}
  \Si{$h_2.\text{fitness}>\min(P).\text{fitness}$}{$\min(P)\leftarrow h_2$}
}
\KwReturn{$\max(P)$}
\end{algorithm}

Se implementan dos variantes: \textbf{AGE-Arit} y \textbf{AGE-BLX}.

\subsection{Algoritmos Meméticos (AM)}

Los AMs hibridan el AGG que mejor resultado ha proporcionado (\textbf{AGG-BLX},
véase Sección~\ref{sec:experimentos}) con la BL de la Práctica~1 adaptada:
fracción de movimiento $p=0{,}15$ (en lugar de $0{,}40$), máximo $100$ evaluaciones
por llamada. La BL se aplica \textbf{cada 10 generaciones}; el presupuesto de cada llamada
es $\min(100,\,\text{maxEvals}-evals)$ para respetar el límite global de $10\,000$
evaluaciones (incluidas las de la BL).

\subsubsection*{Parámetros comunes}
$N_p=100$, $p_c=0{,}8$, $p_m=0{,}1/N$ por gen.

\subsubsection*{AM-All: BL a todos los individuos}
\begin{algorithm}[H]
\caption{AM-All}
\Si{$generaciones \bmod 10 = 0$}{
  \Para{$i\leftarrow 1$ \KwTo $|P|$ \textbf{y} $evals < \text{maxEvals}$}{
    $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
    $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
  }
}
\end{algorithm}

\subsubsection*{AM-Rand: BL aleatoria al 10\% de la población}
\begin{algorithm}[H]
\caption{AM-Rand}
\Si{$generaciones \bmod 10 = 0$}{
  \Para{$i\leftarrow 1$ \KwTo $|P|$ \textbf{y} $evals < \text{maxEvals}$}{
    \Si{rand() $\leq 0{,}1$}{
      $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
      $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
    }
  }
}
\end{algorithm}

\subsubsection*{AM-Best: BL a los $0{,}1\cdot N_p$ mejores}
\begin{algorithm}[H]
\caption{AM-Best}
\Si{$generaciones \bmod 10 = 0$}{
  ordenar $P$ descendente por fitness\;
  \Para{$i\leftarrow 1$ \KwTo $\min(0{,}1\cdot|P|,|P|)$ \textbf{y} $evals < \text{maxEvals}$}{
    $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
    $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
  }
}
\end{algorithm}

\end{center}

BLX-$\alpha$ supera al Aritmético en los tres datasets. El cruce aritmético es
puramente explotativo (genera hijos dentro del convex hull de los padres), mientras que
BLX puede extenderse más allá, permitiendo explorar nuevas regiones. Esta ventaja es
crítica tanto en espacios pequeños (mayor probabilidad de escape de óptimos locales)
como grandes (mayor diversidad necesaria).
Estos resultados justifican que los AMs usen \textbf{AGG-BLX} como base.

\subsection{Comparación de esquemas: AGG vs AGE}

\begin{center}
\begin{tabular}{lcc|c}
\toprule
\textbf{Dataset} & \textbf{AGG-BLX} & \textbf{AGE-BLX} & \textbf{Ganador} \\
\midrule
ibex\_35 & $-4.559$ & $-4.521$ & AGE ($+0.038$) \\
syp\_100 & $-3.074$ & $-2.968$ & AGE ($+0.106$) \\
syp\_500 & $-2.871$ & $-2.876$ & AGG ($+0.005$) \\
\bottomrule
\end{tabular}
\end{center}

AGE supera a AGG en tamaño pequeño-medio; en syp\_500 son prácticamente equivalentes
(diferencia de $0.005$). El AGG reemplaza toda la población cada generación,
provocando \textit{drift} genético pese al elitismo. El AGE mantiene 98 individuos
intactos por iteración, conservando implícitamente una élite. Con presupuesto limitado
(10000 evals), esta estrategia conservadora es superior. En problemas grandes, la
exploración masiva del AGG (50 pares/generación vs.\ 2 en AGE) compensa el drift.

\begin{center}

\textbf{AM-All es el peor AM en todos los datasets}, incluso inferior al mejor AG
(ibex\_35: AM-All $-4.865$ vs AGE-BLX $-4.521$, diferencia $-0.344$).
La causa: AM-All aplica BL a $\approx 90$ individuos en la primera fase memetica

previa. Los individuos inmaduros quedan atrapados en óptimos locales por la BL prematura.

\textbf{AM-Best y AM-Rand} obtienen resultados similares a los mejores AGs, con AM-Best
ligeramente superior (ibex\_35: $-4.503$ vs $-4.516$, diferencia $+0.013$): aplicar
BL a los 10 mejores es más eficiente que seleccionarlos al azar.

\textbf{Análisis de los boxplots} (Figuras~\ref{fig:box_ibex}--\ref{fig:box_syp500}):

\begin{itemize}
  \item \textbf{Greedy}: línea única sin varianza (determinístico) en los tres datasets.
  En ibex\_35 ofrece el mejor valor medio, pero en syp\_100 y syp\_500 es el peor
  algoritmo, lo que refleja que la solución voraz colapsa cuando $N$ crece.

  \item \textbf{Random}: cajas anchas con bigotes largos y outliers frecuentes,
  confirmando la alta varianza. En ibex\_35 aparecen outliers superiores (semillas
  que por azar aciertan buenas combinaciones).

  \item \textbf{BL}: caja muy compacta en todos los datasets, especialmente en ibex\_35
  donde converge siempre en el mismo óptimo local ($\approx 2100$ evals). La
  consistencia aumenta con $N$, confirmando que el operador de movimiento produce
  trayectorias estables.

  \item \textbf{AGs}: varianza baja y homogénea entre variantes, gracias al efecto de
  promediado de la población. BLX tiene medianas ligeramente superiores a Aritmético en
  todos los datasets, y AGE supera a AGG en ibex\_35 y syp\_100.

  \item \textbf{AM-All}: varianza visiblemente mayor que AM-Rand y AM-Best
  (especialmente en syp\_100 y syp\_500), reflejando que el agotamiento prematuro
  del presupuesto hace que el resultado dependa más de la semilla inicial.

  \item \textbf{AM-Rand y AM-Best}: distribuciones similares a los AGs puros en
  varianza, con medianas ligeramente superiores, lo que confirma que la BL selectiva
  mejora la calidad media sin aumentar la variabilidad.
\end{itemize}

\subsection{Conclusiones generales}

\begin{enumerate}
  \item \textbf{BLX supera a Aritmético} en ambos esquemas: la exploración fuera del
  convex hull es crítica.

  \item \textbf{AGE supera a AGG} con presupuesto limitado: la élite implícita es
  más eficiente que el reemplazo generacional.

  \item \textbf{AM-All fracasa} al concentrar la BL en fases tempranas con población
  inmadura. AM-Best y AM-Rand son competitivos pero no dominantes frente a BL.

  \item \textbf{BL domina globalmente}: operador problem-tailored y convergencia
  monotónica son decisivos; el gap respecto a los AGs crece con $N$.

  \item Los \textbf{boxplots} confirman los resultados medios: BL es el más consistente,
  Random el más disperso, y AM-All el AM con mayor varianza.
\end{enumerate}
