\section{Descripción del Problema}
%-----------------------------------------------------------------------

El \textbf{Problema de Portfolio} consiste en determinar la asignación óptima de capital
entre $N$ activos financieros con el objetivo de maximizar el retorno esperado
minimizando simultáneamente el riesgo.

\subsection*{Representación de los datos}

Para cada activo $i \in \{1,\ldots,N\}$ y día $t \in \{1,\ldots,T\}$ se dispone
del retorno diario $\mu_{ti}$, calculado como:
\[
  \mu_{ti} = \frac{V_{ti} - V_{(t+1)i}}{V_{ti}}
\]
Los datos cubren el periodo 2015--2024 para entrenamiento y el año 2025 para evaluación.

\subsection*{Función objetivo}

Dado un vector de pesos $w = (w_1,\ldots,w_N)$:
\[
  \text{fitness}(w) = \text{Beneficio}(w) - \lambda \cdot \text{Riesgo}(w)
\]
\[
  \text{Beneficio}(w) = \sum_{i=1}^{N} w_i \cdot \sum_{t=1}^{T}\log(1+\mu_{ti})
  \qquad
  \text{Riesgo}(w) = \sqrt{\sum_{i=1}^{N}\sum_{j=1}^{N} w_i \cdot \Sigma_{ij} \cdot w_j}
\]
siendo $\Sigma_{ij}$ la covarianza entre activos $i$ y $j$, y $\lambda = 500$.

\begin{algorithm}[H]
\caption{Cálculo de fitness$(w)$}
\KwIn{pesos $w$, beneficio por activo \textit{benefit\_factor}[], matriz de covarianzas \textit{cov}[][], $\lambda$}
\KwOut{valor de fitness}
$benefit \leftarrow 0$\;
\Para{$i \leftarrow 1$ \KwTo $N$}{
  $benefit \leftarrow benefit + w[i] \cdot \textit{benefit\_factor}[i]$\;
}
$risk\_sq \leftarrow 0$\;
\Para{$i \leftarrow 1$ \KwTo $N$}{
  \Para{$j \leftarrow 1$ \KwTo $N$}{
    $risk\_sq \leftarrow risk\_sq + w[i] \cdot \textit{cov}[i][j] \cdot w[j]$\;
  }
}
\KwReturn{$benefit - \lambda \cdot \sqrt{risk\_sq}$}
\end{algorithm}

\subsection*{Restricciones}

\[
  \sum_{i=1}^{N} w_i = 1 \qquad
  w_i = 0 \;\vee\; lo \le w_i \le hi \quad \forall i
\]

\section{Aplicación de los Algoritmos al Problema}
%-----------------------------------------------------------------------

\subsection{Representación de soluciones}

Una solución es un vector $w \in \mathbb{R}^N$ donde $w_i$ indica la fracción de capital
invertida en el activo $i$: exactamente cero o en $[lo, hi]$.

\subsection{Generación aleatoria de soluciones}

Igual que en la Práctica 1: subconjunto aleatorio de activos activos, pesos aleatorios en
$[lo, hi]$, normalización y reparación.

\subsection{Operador de reparación (\texttt{fix})}

Tres fases: normalización, recorte de rango y redistribución iterativa. Es esencial tras
los operadores de cruce, que pueden generar soluciones infactibles.

\subsection{Operador de selección: torneo (k=3)}

Se eligen $k=3$ individuos al azar y se devuelve el de mayor fitness.

\begin{algorithm}[H]
\caption{Torneo ($k=3$)}
\KwIn{población $P$} \KwOut{individuo ganador}
$mejor \leftarrow P[\text{aleatorio en } [0,|P|-1]]$\;
\Para{$i \leftarrow 1$ \KwTo $3$}{
  $rival \leftarrow P[\text{aleatorio en } [0,|P|-1]]$\;
  \Si{$rival.\text{fitness} > mejor.\text{fitness}$}{$mejor \leftarrow rival$\;}
}
\KwReturn{$mejor$}
\end{algorithm}

\subsection{Operadores de cruce}

\subsubsection*{Cruce Aritmético}
Para cada posición $i$ se genera $r_i \in [0,1]$ independiente:
\[
  \text{Off}^1_i = r_i \cdot S^1_i + (1-r_i) \cdot S^2_i
  \qquad
  \text{Off}^2_i = r_i \cdot S^2_i + (1-r_i) \cdot S^1_i
\]

\begin{algorithm}[H]
\caption{Cruce Aritmético($S^1$, $S^2$, problema)}
\KwIn{padres $S^1$, $S^2$; $[lo,hi]$} \KwOut{hijos $\text{Off}^1$, $\text{Off}^2$}
$\text{Off}^1 \leftarrow S^1$;\quad $\text{Off}^2 \leftarrow S^2$\;
\Para{$i \leftarrow 1$ \KwTo $N$}{
  $r \leftarrow$ aleatorio en $[0,1]$\;
  $\text{Off}^1[i] \leftarrow r \cdot S^1[i] + (1-r) \cdot S^2[i]$\;
  $\text{Off}^2[i] \leftarrow r \cdot S^2[i] + (1-r) \cdot S^1[i]$\;
}
\fix{$\text{Off}^1$};\quad \fix{$\text{Off}^2$};\quad \KwReturn{$\text{Off}^1$, $\text{Off}^2$}
\end{algorithm}

\subsubsection*{Cruce BLX-$\alpha$}
Con $I_i = |S^1_i - S^2_i|$:
\[
  hmin_i = \min(S^1_i, S^2_i) - \alpha \cdot I_i \qquad
  hmax_i = \max(S^1_i, S^2_i) + \alpha \cdot I_i
\]
Ambos hijos se generan uniformemente en $[\max(0,hmin_i),\min(hi,hmax_i)]$, $\alpha=0{,}3$.

\begin{algorithm}[H]
\caption{Cruce BLX-$\alpha$($S^1$, $S^2$, problema), $\alpha=0{,}3$}
\KwIn{padres $S^1$, $S^2$; $[lo,hi]$; $\alpha$} \KwOut{hijos $\text{Off}^1$, $\text{Off}^2$}
\Para{$i \leftarrow 1$ \KwTo $N$}{
  $I \leftarrow |S^1[i] - S^2[i]|$\;
  $hmin \leftarrow \max(0,\;\min(S^1[i], S^2[i]) - \alpha \cdot I)$\;
  $hmax \leftarrow \min(hi,\;\max(S^1[i], S^2[i]) + \alpha \cdot I)$\;
  $\text{Off}^1[i] \leftarrow$ aleatorio en $[hmin, hmax]$\;
  $\text{Off}^2[i] \leftarrow$ aleatorio en $[hmin, hmax]$\;
}
\fix{$\text{Off}^1$};\quad \fix{$\text{Off}^2$};\quad \KwReturn{$\text{Off}^1$, $\text{Off}^2$}
\end{algorithm}

\subsection{Operador de mutación}

El operador base es el \textbf{intercambio}: transfiere el 15\% del peso de un gen a otro, preservando $\sum w_i=1$ y los límites $[lo,hi]$ por construcción, sin necesidad de \fix{}.

\begin{algorithm}[H]
\caption{Mutación-intercambio($ind$, $gene\_idx$, problema)}
\KwIn{individuo $ind$, índice $gene\_idx$, $[lo,hi]$}
$gene\_j \leftarrow$ aleatorio en $[0, N-2]$\;
\Si{$gene\_j \geq gene\_idx$}{$gene\_j \leftarrow gene\_j + 1$\;}
$delta \leftarrow 0{,}15 \cdot ind[gene\_idx]$\;
$delta \leftarrow \min(delta,\; ind[gene\_idx] - lo)$\;
$delta \leftarrow \min(delta,\; hi - ind[gene\_j])$\;
\Si{$delta > \varepsilon$}{
  $ind[gene\_idx] \leftarrow ind[gene\_idx] - delta$\;
  $ind[gene\_j] \leftarrow ind[gene\_j] + delta$\;
  evaluar fitness($ind$); $evals \leftarrow evals + 1$\;
}
\end{algorithm}

La forma en que se \textbf{invoca} este operador depende del esquema de evolución, no de si el algoritmo es genético o memético:

\subsubsection*{Esquema estacionario (AGE): probabilístico por individuo}

Con probabilidad $p_m=0{,}1$ por cada hijo generado:

\begin{algorithm}[H]
\caption{Aplicar mutación en AGE($h$, problema)}
\Si{rand() $\leq p_m$}{
  $gene\_idx \leftarrow$ aleatorio en $[0, N-1]$\;
  Mutación-intercambio($h$, $gene\_idx$, problema)\;
}
\end{algorithm}

\subsubsection*{Esquema generacional (AGG y todos los AMs): esperanza matemática}

Se aplica exactamente $n_{mut} = \lfloor p_m \cdot N_p \rfloor = 10$ mutaciones fijas
por generación sobre la nueva población $P'$, tanto en AGG-Arit, AGG-BLX como en
AM-All, AM-Rand y AM-Best (todos comparten el mismo bucle generacional):

\begin{algorithm}[H]
\caption{Aplicar mutaciones en AGG/AMs (esperanza matemática)}
$n_{mut} \leftarrow \lfloor p_m \cdot N_p \rfloor$\;
$pop\_actual \leftarrow |P'|$\;
\Para{$k \leftarrow 1$ \KwTo $n_{mut}$ \textbf{y} $evals < \text{maxEvals}$}{
  $chrom\_idx \leftarrow$ aleatorio en $[0,\, pop\_actual - 1]$\;
  $gene\_idx \leftarrow$ aleatorio en $[0, N-1]$\;
  Mutación-intercambio($P'[chrom\_idx]$, $gene\_idx$, problema)\;
}
\end{algorithm}

\subsection{Búsqueda Local corregida (usada en los AMs)}

En la Práctica 1, la BL aplicaba \fix{} sobre cada solución candidata, lo cual era
subóptimo dado que el propio operador de movimiento podría preservar las restricciones.
En la Práctica 2, la BL utiliza un \textbf{operador de intercambio} que preserva
$\sum w_i = 1$ y los límites $[lo,hi]$ por construcción, eliminando la necesidad de
\fix{} en cada paso.

El operador de intercambio funciona del siguiente modo: se selecciona un par aleatorio
de índices (origen, destino), se calcula el mayor desplazamiento $\delta$ compatible
con los límites, y si $\delta > 0$ se transfiere esa cantidad de peso de origen a
destino.

\begin{algorithm}[H]
\caption{BL($sol\_inicial$, problema, maxEvals, $p=0{,}15$)}
\KwIn{solución inicial, problema, maxEvals, proporción $p$}
\KwOut{solución, fitness, evals}
$Pares \leftarrow \{(i,j) : i \neq j,\; i,j \in [0,N-1]\}$\quad \textit{// $N(N-1)$ pares}\;
$sol \leftarrow sol\_inicial$\;
$fitness \leftarrow$ evaluar($sol$); $evals \leftarrow 1$\;
$mejora \leftarrow$ verdadero\;
\Mientras{$mejora$ \textbf{y} $evals < \text{maxEvals}$}{
  $mejora \leftarrow$ falso\;
  Barajar $Pares$ aleatoriamente\;
  \Para{cada $(origen, destino)$ en $Pares$ \textbf{y} $evals < \text{maxEvals}$}{
    $delta \leftarrow p \cdot sol[origen]$\;
    $delta \leftarrow \min(delta,\; sol[origen] - lo)$\;
    $delta \leftarrow \min(delta,\; hi - sol[destino])$\;
    \Si{$delta \leq \varepsilon$}{\textbf{continuar}\;}
    $sol' \leftarrow sol$;\quad $sol'[origen] \mathrel{-}= delta$;\quad $sol'[destino] \mathrel{+}= delta$\;
    $fit' \leftarrow$ evaluar($sol'$); $evals \leftarrow evals + 1$\;
    \Si{$fit' > fitness$}{
      $sol \leftarrow sol'$; $fitness \leftarrow fit'$\;
      $mejora \leftarrow$ verdadero; \textbf{romper}\;
    }
  }
}
\KwReturn{$sol$, $fitness$, $evals$}
\end{algorithm}

\subsection{Algoritmo Genético Generacional (AGG)}

\subsubsection*{Parámetros}
$N_p=100$, $p_c=0{,}8$, $p_m=0{,}1$ (equivale a $n_c=40$ cruces y $n_m=10$ mutaciones fijas/generación), criterio de parada: $10\,000$ evaluaciones.

\subsubsection*{Esquema de evolución y reemplazamiento}

El AGG usa la \textbf{esperanza matemática} tanto para el cruce como para la mutación:
se calculan exactamente $n_{cruces} = \lfloor p_c \cdot N_p / 2 \rfloor = 40$ pares que
se cruzan, y los $N_p/2 - 40 = 10$ pares restantes producen copias directas. La mutación
aplica exactamente $n_{mut} = \lfloor p_m \cdot N_p \rfloor = 10$ intercambios a
individuos elegidos aleatoriamente.

\begin{algorithm}[H]
\caption{AGG (generacional con elitismo, esperanza matemática)}
\KwIn{problema, maxEvals} \KwOut{mejor solución}
Inicializar $P$ con $N_p$ aleatorios; $evals \leftarrow N_p$\;
\Mientras{$evals < \text{maxEvals}$}{
  $padres \leftarrow [$ \torneo{$P$} $\times N_p]$;\quad $n_c \leftarrow \lfloor p_c \cdot N_p/2 \rfloor$;\quad $P' \leftarrow \emptyset$\;
  \Para{$i \leftarrow 0$ \KwTo $N_p/2-1$ \textbf{y} $evals<\text{maxEvals}$}{
    $p_1 \leftarrow padres[2i]$;\quad $p_2 \leftarrow padres[2i{+}1]$\;
    \Si{$i < n_c$}{$(h_1,h_2)\leftarrow \mathrm{cruzar}(p_1,p_2)$;\quad $evals\leftarrow evals+2$\;}
    \OSiNo{$h_1\leftarrow p_1$;\quad $h_2\leftarrow p_2$\;}
    Añadir $h_1,h_2$ a $P'$\;
  }
  $n_m \leftarrow \lfloor p_m \cdot N_p \rfloor$;\quad
  \Para{$k \leftarrow 0$ \KwTo $n_m-1$ \textbf{y} $evals<\text{maxEvals}$}{
    $chr\leftarrow$ rand $[0,|P'|{-}1]$;\quad $g\leftarrow$ rand $[0,N{-}1]$;\quad Mutación-intercambio($P'[chr],g$)\;
  }
  \Si{$\max(P').fit < \max(P).fit$}{$\min(P') \leftarrow \max(P)$\;}
  $P \leftarrow P'$\;
}
\KwReturn{$\max(P)$}
\end{algorithm}

Se implementan dos variantes: \textbf{AGG-Arit} y \textbf{AGG-BLX}.

\subsection{Algoritmo Genético Estacionario (AGE)}

\subsubsection*{Parámetros}
$N_p=100$, $p_c=1{,}0$ (siempre se cruzan), $p_m=0{,}1$ por individuo,
criterio de parada: $10\,000$ evaluaciones.

\subsubsection*{Esquema de evolución y reemplazamiento}

\begin{algorithm}[H]
\caption{AGE (estacionario)}
\KwIn{problema, maxEvals} \KwOut{mejor solución}
Inicializar $P$ con $N_p$ aleatorios; $evals \leftarrow N_p$\;
\Mientras{$evals < \text{maxEvals}$}{
  $p_1\leftarrow$\torneo{$P$};\quad $p_2\leftarrow$\torneo{$P$}\;
  \Si{rand() $\leq p_c$}{$(h_1,h_2)\leftarrow \mathrm{cruzar}(p_1,p_2)$;\quad $evals\leftarrow evals+2$\;}
  \OSiNo{$h_1\leftarrow p_1$;\quad $h_2\leftarrow p_2$\;}
  \Si{rand() $\leq p_m$}{$g\leftarrow$rand$[0,N{-}1]$;\quad Mutación-intercambio($h_1,g$)\;}
  \Si{rand() $\leq p_m$}{$g\leftarrow$rand$[0,N{-}1]$;\quad Mutación-intercambio($h_2,g$)\;}
  $peor_1\leftarrow\min(P)$\;
  \Si{$h_1.fit > peor_1.fit$}{$peor_1\leftarrow h_1$\;}
  $peor_2\leftarrow\min(P)$\;
  \Si{$h_2.fit > peor_2.fit$}{$peor_2\leftarrow h_2$\;}
}
\KwReturn{$\max(P)$}
\end{algorithm}

\textit{El $\min(P)$ se recalcula para $h_2$ después de la posible inserción de $h_1$.}

Se implementan dos variantes: \textbf{AGE-Arit} y \textbf{AGE-BLX}.

\subsection{Algoritmos Meméticos (AM)}

Los AMs hibridan el AGG que mejor resultado ha proporcionado (\textbf{AGG-BLX},
véase Sección~\ref{sec:experimentos}) con la BL de la Práctica~1 adaptada:
fracción de movimiento $p=0{,}15$ (en lugar de $0{,}40$), máximo $100$ evaluaciones
por llamada. La BL se aplica \textbf{cada 10 generaciones}; el presupuesto de cada llamada
es $\min(100,\,\text{maxEvals}-evals)$ para respetar el límite global de $10\,000$
evaluaciones (incluidas las de la BL).

Los AMs comparten \textbf{exactamente el mismo bucle generacional que el AGG}: misma
selección ($N_p$ torneos, $k=3$), mismo cruce con esperanza matemática ($n_c=40$ pares fijos),
mismo emparejamiento fijo, misma mutación con esperanza matemática ($n_m=10$ fijos) y
mismo elitismo. La única diferencia estructural es el bloque de BL que se activa
cada 10 generaciones según el criterio de cada variante.

\subsubsection*{Parámetros comunes}
$N_p=100$, $p_c=0{,}8$, $p_m=0{,}1/N$ por gen (equivale a $n_m=10$ mutaciones/generación).

\subsubsection*{AM-All: BL a todos los individuos}
\begin{algorithm}[H]
\caption{AM-All}
\Si{$generaciones \bmod 10 = 0$}{
  \Para{$i\leftarrow 1$ \KwTo $|P|$ \textbf{y} $evals < \text{maxEvals}$}{
    $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
    $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
  }
}
\end{algorithm}

\subsubsection*{AM-Rand: BL aleatoria al 10\% de la población}
\begin{algorithm}[H]
\caption{AM-Rand}
\Si{$generaciones \bmod 10 = 0$}{
  \Para{$i\leftarrow 1$ \KwTo $|P|$ \textbf{y} $evals < \text{maxEvals}$}{
    \Si{rand() $\leq 0{,}1$}{
      $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
      $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
    }
  }
}
\end{algorithm}

\subsubsection*{AM-Best: BL a los $0{,}1\cdot N_p$ mejores}
\begin{algorithm}[H]
\caption{AM-Best}
\Si{$generaciones \bmod 10 = 0$}{
  ordenar $P$ descendente por fitness\;
  \Para{$i\leftarrow 1$ \KwTo $\min(0{,}1\cdot|P|,|P|)$ \textbf{y} $evals < \text{maxEvals}$}{
    $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
    $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
  }
}
\end{algorithm}

\subsection{BLX vs Aritmético}

BLX-$\alpha$ supera al cruce Aritmético en las \textbf{6 comparaciones} (3 datasets
$\times$ 2 esquemas AGG y AGE), sin excepción.

\begin{center}
\begin{tabular}{lcccc}
\toprule
\textbf{Dataset} & \textbf{AGG-Arit} & \textbf{AGG-BLX} & \textbf{AGE-Arit} & \textbf{AGE-BLX} \\
\midrule
ibex\_35 & $-4.644$ & $-4.525\;(+0.119)$ & $-4.614$ & $-4.520\;(+0.094)$ \\
syp\_100 & $-3.112$ & $-3.035\;(+0.077)$ & $-3.158$ & $-3.049\;(+0.109)$ \\
syp\_500 & $-2.961$ & $-2.877\;(+0.084)$ & $-3.201$ & $-3.095\;(+0.106)$ \\
\bottomrule
\end{tabular}
\end{center}

El cruce Aritmético es puramente explotativo: genera hijos dentro del \textit{convex hull}
de los padres. BLX puede extenderse más allá de ese convex hull (controlado por $\alpha$),
permitiendo explorar nuevas regiones del espacio de soluciones. Esta capacidad de
exploración resulta crítica tanto en espacios pequeños (escape de óptimos locales) como
en espacios de alta dimensión (diversidad necesaria). Estos resultados justifican que
los AMs usen \textbf{AGG-BLX} como base.

\subsection{AGG vs AGE: influencia del tamaño del problema}

\begin{center}
\begin{tabular}{lcc|c}
\toprule
\textbf{Dataset} & \textbf{AGG-BLX} & \textbf{AGE-BLX} & \textbf{Ganador} \\
\midrule
ibex\_35  & $-4.525$ & $-4.520$ & AGE ($+0.005$) \\
syp\_100  & $-3.035$ & $-3.049$ & AGG ($+0.014$) \\
syp\_500  & $-2.877$ & $-3.095$ & AGG ($+0.218$) \\
\bottomrule
\end{tabular}
\end{center}

El resultado más relevante es el de syp\_500: \textbf{AGG supera a AGE por un margen de
$0.218$}, mientras que en los datasets pequeños las diferencias son mínimas.

La explicación reside en la dimensionalidad: en syp\_500 ($N=457$), cada generación del
AGG genera 100 nuevos individuos que exploran el espacio de búsqueda de forma diversa,
cubriendo una mayor región del espacio en cada iteración. El AGE, que solo produce 2
hijos por iteración, tiene un ritmo de exploración muy inferior en espacios de alta
dimensión. Para ibex\_35 ($N=30$), el elitismo implícito del AGE (98 individuos intactos
por iteración) es más valioso que la exploración masiva del AGG con presupuesto limitado.

\subsection{Algoritmos Meméticos: resultados y análisis}

\textbf{AM-Best y AM-Rand son los mejores algoritmos en todos los datasets}, superando
incluso a la Búsqueda Local:

\begin{center}
\begin{tabular}{lrrr}
\toprule
\textbf{Dataset} & \textbf{BL} & \textbf{AM-Rand} & \textbf{AM-Best} \\
\midrule
ibex\_35 & $-4.573$ & $-4.491$ ($+0.082$) & $-4.491$ ($+0.082$) \\
syp\_100 & $-3.032$ & $-2.941$ ($+0.091$) & $-2.912$ ($+0.120$) \\
syp\_500 & $-3.318$ & $-2.846$ ($+0.472$) & $-2.811$ ($+0.507$) \\
\bottomrule
\end{tabular}
\end{center}

La ventaja crece con $N$: en syp\_500, AM-Best supera a BL en $0.507$ unidades de
fitness. El enfoque híbrido combina la exploración global de los AGs con la explotación
local de la BL aplicada a los mejores individuos.

\textbf{AM-All es el peor AM en todos los datasets.} La causa es el agotamiento
prematuro del presupuesto: AM-All aplica BL a los 100 individuos cada 10 generaciones,
lo que supone $100 \times 100 = 10\,000$ evaluaciones en la primera aplicación (en la
generación 10), agotando prácticamente todo el presupuesto. Apenas quedan evaluaciones
para la evolución posterior. Los individuos, todavía inmaduros en la generación 10,
quedan atrapados en óptimos locales por la BL prematura.

\subsection{Análisis de la Búsqueda Local}

Con los nuevos datos, la BL ya no es dominante. El cambio más notable es en syp\_500:
la BL alcanza $-3.318$ de media usando casi todo el presupuesto ($\approx 9840$ evals),
mientras que AGG-BLX logra $-2.877$ y AM-Best $-2.811$.

La razón es la \textbf{explosión del vecindario}: en syp\_500 ($N=457$), el número de
pares posibles para el operador de intercambio es $457 \times 456 = 208\,392$. Con un
límite de $\approx 9840$ evaluaciones, la BL solo puede explorar menos del $5\%$ de una
única pasada completa por el vecindario. En estas condiciones, la BL queda atrapada en
el primer óptimo local que encuentra, sin capacidad de exploración global.

Los AGs y AMs, en cambio, mantienen una población diversa que explora múltiples regiones
del espacio simultáneamente, siendo mucho más efectivos para $N$ grande.

\subsection{Análisis de la variabilidad ($\sigma$) en AGs y AMs}

Los algoritmos de esta práctica (AGs y AMs) muestran patrones de variabilidad
muy distintos entre sí, directamente relacionados con su diseño:

\subsubsection*{Variabilidad entre operadores de cruce}

El cruce BLX produce consistentemente \textbf{menor varianza que el Aritmético}
en ambos esquemas y los tres datasets:

\begin{center}\footnotesize
\begin{tabular}{lcc|cc|cc}
\toprule
 & \multicolumn{2}{c|}{\textbf{ibex\_35}} & \multicolumn{2}{c|}{\textbf{syp\_100}} & \multicolumn{2}{c}{\textbf{syp\_500}} \\
 & Arit $\sigma$ & BLX $\sigma$ & Arit $\sigma$ & BLX $\sigma$ & Arit $\sigma$ & BLX $\sigma$ \\
\midrule
AGG & $0.050$ & $0.018$ & $0.040$ & $0.038$ & $0.073$ & $0.043$ \\
AGE & $0.036$ & $0.012$ & $0.054$ & $0.042$ & $0.069$ & $0.044$ \\
\bottomrule
\end{tabular}
\end{center}

BLX, al explorar más allá del convex hull de los padres, guía la búsqueda hacia
regiones más prometedoras de forma más consistente, reduciendo la sensibilidad a la
semilla inicial. El cruce Aritmético, puramente explotativo, queda más atrapado
en la cuenca de atracción de la solución inicial.

\subsubsection*{Variabilidad entre esquemas: AGG vs AGE}

AGE presenta \textbf{menor $\sigma$ que AGG} en ibex\_35 y syp\_100 para ambos
operadores de cruce. El esquema estacionario, al reemplazar solo 0--2 individuos por
iteración, mantiene una élite implícita grande que estabiliza la búsqueda. AGG, al
regenerar toda la población cada generación, introduce mayor variabilidad generación
a generación. En syp\_500 la diferencia de $\sigma$ es mínima entre AGG y AGE,
dominando el efecto de la dimensionalidad.

\subsubsection*{Variabilidad de los AMs: efecto de la estrategia de BL}

Es en los AMs donde la $\sigma$ muestra las diferencias más extremas:

\begin{itemize}
  \item \textbf{AM-Rand y AM-Best}: $\sigma \approx 0.001$ en ibex\_35 — el resultado
  más llamativo de toda la práctica. Todas las semillas convergen al mismo óptimo
  independientemente del punto de inicio. La BL selectiva (aplicada a 10 individuos
  ya maduros evolutivamente) combina exploración global del AGG con explotación local
  precisa, encontrando siempre el mismo atractor. En syp\_100 y syp\_500 la varianza
  sube ($\sigma \approx 0.03$--$0.04$) pero sigue siendo la más baja de todos los
  algoritmos evolutivos.

  \item \textbf{AM-All}: paradójicamente, tiene la \textbf{mayor $\sigma$ de los AMs}
  en syp\_100 ($0.045$) y syp\_500 ($0.040$), mayor incluso que algunos AGs puros. Al
  aplicar BL a los 100 individuos en generación 10 — cuando la población aún está
  inmadura —, el resultado queda fuertemente ligado a la calidad de esa población inicial,
  que varía mucho según la semilla. Los AGs puros son más estables porque no dependen de
  una fase de BL temprana.
\end{itemize}

ya que ambos aplican BL a $\approx$10 individuos por fase memetica.

\subsection{Análisis de los boxplots}

Los boxplots (Figuras~\ref{fig:box_ibex}--\ref{fig:box_syp500}) permiten observar
tanto la calidad media como la consistencia de cada algoritmo:

\begin{itemize}
  \item \textbf{Greedy}: línea única sin caja (completamente determinístico). En
  ibex\_35 su valor es competitivo ($-4.650$), pero en syp\_100 y syp\_500 aparece
  cortado en la parte inferior del gráfico, siendo el peor de todos: la heurística voraz
  colapsa en espacios de alta dimensión.

  \item \textbf{Random}: caja amplia con bigotes largos y outliers en los tres datasets,
  confirmando su alta sensibilidad a la semilla. En ibex\_35 aparecen outliers superiores
  (semillas que por azar generan buenas combinaciones iniciales).

  \item \textbf{BL}: la \textbf{Búsqueda Local empleada en esta práctica es la versión
  corregida respecto a la Práctica~1}: el operador de intercambio preserva las
  restricciones por construcción, eliminando las llamadas innecesarias a \texttt{fix()}
  que distorsionaban el vecindario. Esta corrección es visible en la consistencia de los
  resultados.

  En \textbf{ibex\_35} la caja es la más compacta del gráfico ($\sigma=0.019$): BL
  converge siempre al mismo óptimo local en solo $\approx$1\,257 evals, lo que confirma
  un paisaje de fitness suave con un atractor dominante. En \textbf{syp\_100} la caja es
  moderada ($\sigma=0.067$, converge en $\approx$5\,258 evals) y la mediana es
  competitiva con los AGs. En \textbf{syp\_500} ocurre lo contrario: \textbf{BL tiene
  la caja más ancha de todos los algoritmos} ($\sigma=0.164$). La causa es estructural:
  con $N=457$ el vecindario tiene $457\times456=208{.}392$ pares; con 10\,000
  evaluaciones BL solo puede explorar el $4{,}8\%$ de una única pasada, parando siempre
  por presupuesto ($\approx$9\,840 evals) y nunca por convergencia. El resultado depende
  de qué fracción aleatoria del vecindario exploró cada semilla, produciendo alta
  varianza. Los AGs tienen mucha menor varianza porque la población distribuye el
  presupuesto de forma intrínsecamente más robusta.

  \item \textbf{AGs}: varianza reducida y homogénea en todos los datasets. Las
  diferencias entre variantes son visibles: \textbf{BLX tiene medianas superiores a
  Aritmético} tanto en AGG como en AGE, confirmando la ventaja exploradora de BLX. En
  ibex\_35 AGE-BLX supera ligeramente a AGG-BLX; en syp\_500 la diferencia
  AGG-BLX\,$-$\,AGE-BLX es la más visible del gráfico ($\approx0{,}22$ de fitness),
  mostrando que el esquema generacional escala mejor con $N$ grande.

  \item \textbf{AM-All}: en ibex\_35 su caja es la más grande de los AMs con un
  outlier hacia $-4{,}89$; en syp\_100 es incluso \textbf{peor en mediana que todos los
  AGs puros} con varianza considerable. Esto refleja el agotamiento prematuro del
  presupuesto: aplicar BL a los 100 individuos en la primera fase memetica (generación
  10) consume $\approx$10\,000 evals de golpe, dejando sin presupuesto de evolución.
  La alta varianza indica que el resultado queda muy ligado a la calidad de la población
  en generación 10, que depende mucho de la semilla.

  \item \textbf{AM-Rand y AM-Best}: en ibex\_35 sus cajas son \textbf{prácticamente
  invisibles} ($\sigma\approx0{,}001$), con medianas claramente superiores a todos los
  demás algoritmos incluida la BL. En syp\_100 y syp\_500 mantienen varianza baja y
  medianas superiores. AM-Best tiene medianas ligeramente superiores a AM-Rand en todos
  los datasets, confirmando que dirigir la BL a los mejores individuos es más eficiente
  que aplicarla aleatoriamente.
\end{itemize}

\subsection{Capacidad de generalización (test 2025)}

Los rankings de entrenamiento se mantienen en test para los algoritmos más robustos.
AM-Best generaliza bien en todos los datasets: ibex\_35 test $-3.084$ (mejor de todos),
syp\_100 test $-3.725$ (mejor), syp\_500 test $-3.907$ (mejor). AGG-BLX y AGE-BLX
también generalizan correctamente, con test próximos a los de entrenamiento.

El caso más llamativo es \textbf{Greedy en syp\_100 y syp\_500}: el fitness de test
($-5.020$ y $-5.205$) es muy inferior al de entrenamiento ($-3.664$ y $-3.438$),
indicando que la solución voraz sobreajusta al periodo histórico y no generaliza.

\subsection{Conclusiones generales}

\begin{enumerate}
  \item \textbf{BLX supera a Aritmético} en las 6 comparaciones (3 datasets $\times$
  2 esquemas): la exploración fuera del convex hull es crítica para la calidad.

  \item \textbf{AGE supera a AGG} para $N$ pequeño (ibex\_35); \textbf{AGG supera a AGE}
  para $N$ grande (syp\_500, margen $0.218$). La exploración masiva del AGG es decisiva
  en alta dimensión.

  \item \textbf{AM-Best y AM-Rand son los mejores algoritmos globales}, superando a BL
  en todos los datasets. La hibridación evolutivo-local es la estrategia ganadora.

  \item \textbf{AM-All fracasa} por agotamiento del presupuesto en la generación 10:
  $100 \times 100 = 10\,000$ evaluaciones de BL consumen todo el presupuesto sin
  evolución suficiente previa.

  \item \textbf{BL no es competitiva para $N$ grande}: la explosión del vecindario
  ($457 \times 456 = 208\,392$ pares en syp\_500) impide explorar más del $5\%$ del
  vecindario con el presupuesto disponible.
\end{enumerate}

La Figura~\ref{fig:blx_alpha_curve} muestra la mejora porcentual del fitness respecto
a $\alpha=0$ en función de $\alpha$ para los tres datasets. La forma de \textbf{U
invertida} es visible en los tres casos, aunque con perfiles distintos:

\begin{itemize}
  \item \textbf{IBEX 35 (azul, $N=30$)}: curva muy plana. La mejora máxima es solo del
  $3{,}5\%$ y la curva hace plateau en $\alpha \in [0.5, 0.7]$ con apenas descenso en
  $\alpha=1.0$. Con un espacio de soluciones pequeño ($N=30$), incluso $\alpha$ bajos
  producen exploración suficiente; valores altos no degradan significativamente la calidad.

  \item \textbf{S\&P 100 (rojo, $N=97$)}: U invertida más pronunciada y clásica. El
  pico es del $9{,}5\%$ en $\alpha=0.5$ y la caída a $\alpha=1.0$ es muy marcada (solo
  $2{,}2\%$ de mejora restante). Es el dataset donde la elección de $\alpha$ tiene mayor
  impacto relativo.

  \item \textbf{S\&P 500 (verde, $N=457$)}: pico más alto ($10{,}5\%$) pero la
  bajada posterior es más suave ($6{,}9\%$ aún en $\alpha=1.0$). El gran espacio de
  soluciones aguanta más exploración antes de que sea perjudicial; la zona óptima es
  más ancha.
\end{itemize}

\textbf{$\alpha = 0.5$ es óptimo en los tres datasets}. El valor $\alpha = 0.3$
recomendado en el guion es subóptimo: $\alpha = 0.5$ mejora el fitness en
$\approx 7\%$ en syp\_100 ($-3.035$ vs $-2.834$) y syp\_500 ($-2.877$ vs $-2.682$).

Un hallazgo contraintuitivo es que \textbf{$\sigma$ disminuye al aumentar $\alpha$}
hasta el óptimo: más exploración produce resultados más consistentes entre semillas.
Con $\alpha$ bajo la población converge a distintos óptimos locales según la semilla
(alta varianza); con $\alpha$ adecuado escapa de esos óptimos y converge siempre
hacia el mismo atractor. El caso extremo es ibex\_35 con $\alpha\in\{0.5, 0.7\}$:
$\sigma=0.000$, prácticamente determinístico en las 50 semillas.

Finalmente, $\alpha=0.0$ reproduce el comportamiento del cruce Aritmético (sin
extensión del intervalo), confirmando que la ventaja de BLX proviene exclusivamente
de la exploración más allá del convex hull de los padres.
