\section{Descripción del Problema}
%-----------------------------------------------------------------------

El \textbf{Problema de Portfolio} consiste en determinar la asignación óptima de capital
entre $N$ activos financieros con el objetivo de maximizar el retorno esperado
minimizando simultáneamente el riesgo.

\subsection*{Representación de los datos}

Para cada activo $i \in \{1,\ldots,N\}$ y día $t \in \{1,\ldots,T\}$ se dispone
del retorno diario $\mu_{ti}$, calculado como:
\[
  \mu_{ti} = \frac{V_{ti} - V_{(t+1)i}}{V_{ti}}
\]
Los datos cubren el periodo 2015--2024 para entrenamiento y el año 2025 para evaluación.

\subsection*{Función objetivo}

Dado un vector de pesos $w = (w_1,\ldots,w_N)$:
\[
  \text{fitness}(w) = \text{Beneficio}(w) - \lambda \cdot \text{Riesgo}(w)
\]
\[
  \text{Beneficio}(w) = \sum_{i=1}^{N} w_i \cdot \sum_{t=1}^{T}\log(1+\mu_{ti})
  \qquad
  \text{Riesgo}(w) = \sqrt{\sum_{i=1}^{N}\sum_{j=1}^{N} w_i \cdot \Sigma_{ij} \cdot w_j}
\]
siendo $\Sigma_{ij}$ la covarianza entre activos $i$ y $j$, y $\lambda = 500$.

\begin{algorithm}[H]
\caption{Cálculo de fitness$(w)$}
\KwIn{pesos $w$, beneficio por activo \textit{benefit\_factor}[], matriz de covarianzas \textit{cov}[][], $\lambda$}
\KwOut{valor de fitness}
$benefit \leftarrow 0$\;
\Para{$i \leftarrow 1$ \KwTo $N$}{
  $benefit \leftarrow benefit + w[i] \cdot \textit{benefit\_factor}[i]$\;
}
$risk\_sq \leftarrow 0$\;
\Para{$i \leftarrow 1$ \KwTo $N$}{
  \Para{$j \leftarrow 1$ \KwTo $N$}{
    $risk\_sq \leftarrow risk\_sq + w[i] \cdot \textit{cov}[i][j] \cdot w[j]$\;
  }
}
\KwReturn{$benefit - \lambda \cdot \sqrt{risk\_sq}$}
\end{algorithm}

\subsection*{Restricciones}

\[
  \sum_{i=1}^{N} w_i = 1 \qquad
  w_i = 0 \;\vee\; lo \le w_i \le hi \quad \forall i
\]

\section{Aplicación de los Algoritmos al Problema}
%-----------------------------------------------------------------------

\subsection{Representación de soluciones}

Una solución es un vector $w \in \mathbb{R}^N$ donde $w_i$ indica la fracción de capital
invertida en el activo $i$: exactamente cero o en $[lo, hi]$.

\subsection{Generación aleatoria de soluciones}

Igual que en la Práctica 1: subconjunto aleatorio de activos activos, pesos aleatorios en
$[lo, hi]$, normalización y reparación.

\subsection{Operador de reparación (\texttt{fix})}

Tres fases: normalización, recorte de rango y redistribución iterativa. Es esencial tras
los operadores de cruce, que pueden generar soluciones infactibles.

\subsection{Operador de selección: torneo (k=3)}

Se eligen $k=3$ individuos al azar y se devuelve el de mayor fitness.

\begin{algorithm}[H]
\caption{Torneo ($k=3$)}
\KwIn{población $P$} \KwOut{individuo ganador}
$i_1 \leftarrow$ aleatorio en $[0,|P|-1]$\;
$i_2 \leftarrow$ aleatorio en $[0,|P|-1]$\;
$i_3 \leftarrow$ aleatorio en $[0,|P|-1]$\;
\KwReturn{$P[\arg\max_{k\in\{i_1,i_2,i_3\}} P[k].\text{fitness}]$}
\end{algorithm}

\subsection{Operadores de cruce}

\subsubsection*{Cruce Aritmético}
Para cada posición $i$ se genera $r_i \in [0,1]$ independiente:
\[
  \text{Off}^1_i = r_i \cdot S^1_i + (1-r_i) \cdot S^2_i
  \qquad
  \text{Off}^2_i = r_i \cdot S^2_i + (1-r_i) \cdot S^1_i
\]

\begin{algorithm}[H]
\caption{Cruce Aritmético($S^1$, $S^2$, problema)}
\KwIn{padres $S^1$, $S^2$; $[lo,hi]$} \KwOut{hijos $\text{Off}^1$, $\text{Off}^2$}
$\text{Off}^1 \leftarrow S^1$;\quad $\text{Off}^2 \leftarrow S^2$\;
\Para{$i \leftarrow 1$ \KwTo $N$}{
  $r \leftarrow$ aleatorio en $[0,1]$\;
  $\text{Off}^1[i] \leftarrow r \cdot S^1[i] + (1-r) \cdot S^2[i]$\;
  $\text{Off}^2[i] \leftarrow r \cdot S^2[i] + (1-r) \cdot S^1[i]$\;
}
\fix{$\text{Off}^1$};\quad \fix{$\text{Off}^2$};\quad \KwReturn{$\text{Off}^1$, $\text{Off}^2$}
\end{algorithm}

\subsubsection*{Cruce BLX-$\alpha$}
Con $I_i = |S^1_i - S^2_i|$:
\[
  hmin_i = \min(S^1_i, S^2_i) - \alpha \cdot I_i \qquad
  hmax_i = \max(S^1_i, S^2_i) + \alpha \cdot I_i
\]
Ambos hijos se generan uniformemente en $[\max(0,hmin_i),\min(hi,hmax_i)]$, $\alpha=0{,}3$.

\begin{algorithm}[H]
\caption{Cruce BLX-$\alpha$($S^1$, $S^2$, problema), $\alpha=0{,}3$}
\KwIn{padres $S^1$, $S^2$; $[lo,hi]$; $\alpha$} \KwOut{hijos $\text{Off}^1$, $\text{Off}^2$}
\Para{$i \leftarrow 1$ \KwTo $N$}{
  $I \leftarrow |S^1[i] - S^2[i]|$\;
  $hmin \leftarrow \max(0,\;\min(S^1[i], S^2[i]) - \alpha \cdot I)$\;
  $hmax \leftarrow \min(hi,\;\max(S^1[i], S^2[i]) + \alpha \cdot I)$\;
  $\text{Off}^1[i] \leftarrow$ aleatorio en $[hmin, hmax]$\;
  $\text{Off}^2[i] \leftarrow$ aleatorio en $[hmin, hmax]$\;
}
\fix{$\text{Off}^1$};\quad \fix{$\text{Off}^2$};\quad \KwReturn{$\text{Off}^1$, $\text{Off}^2$}
\end{algorithm}

\subsection{Operador de mutación}

\subsubsection*{Para Algoritmos Genéticos}

El operador de mutación para los AGs usa un \textbf{intercambio} entre dos genes, lo que
preserva la restricción $\sum w_i = 1$ y los límites $[lo,hi]$ por construcción, sin
necesidad de \fix{}.
Con probabilidad $p_m=0{,}1$ por individuo se aplica la mutación de intercambio.

\begin{algorithm}[H]
\caption{Mutación AG($ind$, problema)}
\Si{rand() $\leq 0{,}1$}{
  $gene\_idx \leftarrow$ aleatorio en $[0, N-1]$\;
  $gene\_j \leftarrow$ aleatorio en $[0, N-1]$, $gene\_j \neq gene\_idx$\;
  $delta \leftarrow 0{,}15 \cdot w[gene\_idx]$\;
  $delta \leftarrow \min(delta,\; w[gene\_idx] - lo)$\;
  $delta \leftarrow \min(delta,\; hi - w[gene\_j])$\;
  \Si{$delta > 0$}{
    $w[gene\_idx] \leftarrow w[gene\_idx] - delta$\;
    $w[gene\_j] \leftarrow w[gene\_j] + delta$\;
    evaluar fitness; $evals \leftarrow evals + 1$\;
  }
}
\end{algorithm}

\subsubsection*{Para Algoritmos Meméticos}

En los AMs (basados en AGG), se usa la \textbf{esperanza matemática}: en lugar de
aplicar la mutación con probabilidad $p_m$ por individuo, se calculan exactamente
$n_{mut} = \lfloor p_m \cdot N_p \rfloor = \lfloor 0{,}1 \cdot 100 \rfloor = 10$
mutaciones de intercambio por generación, eligiendo aleatoriamente los pares
(cromosoma, gen) a mutar.

\begin{algorithm}[H]
\caption{Mutación AM (esperanza matemática)}
$n_{mut} \leftarrow \lfloor 0{,}1 \cdot N_p \rfloor$\;
\Para{$k \leftarrow 1$ \KwTo $n_{mut}$}{
  $chr\_idx \leftarrow$ aleatorio en $[0, N_p-1]$\;
  $gene\_idx \leftarrow$ aleatorio en $[0, N-1]$\;
  $gene\_j \leftarrow$ aleatorio en $[0, N-1]$, $gene\_j \neq gene\_idx$\;
  $delta \leftarrow 0{,}15 \cdot P[chr\_idx][gene\_idx]$\;
  $delta \leftarrow \min(delta,\; P[chr\_idx][gene\_idx] - lo)$\;
  $delta \leftarrow \min(delta,\; hi - P[chr\_idx][gene\_j])$\;
  \Si{$delta > 0$}{
    aplicar intercambio; evaluar fitness; $evals \leftarrow evals + 1$\;
  }
}
\end{algorithm}

\subsection{Búsqueda Local corregida (usada en los AMs)}

En la Práctica 1, la BL aplicaba \fix{} sobre cada solución candidata, lo cual era
subóptimo dado que el propio operador de movimiento podría preservar las restricciones.
En la Práctica 2, la BL utiliza un \textbf{operador de intercambio} que preserva
$\sum w_i = 1$ y los límites $[lo,hi]$ por construcción, eliminando la necesidad de
\fix{} en cada paso.

El operador de intercambio funciona del siguiente modo: se selecciona un par aleatorio
de índices (origen, destino), se calcula el mayor desplazamiento $\delta$ compatible
con los límites, y si $\delta > 0$ se transfiere esa cantidad de peso de origen a
destino.

\begin{algorithm}[H]
\caption{Paso de intercambio en la BL (Práctica 2)}
\KwIn{solución $sol$, proporción $p=0{,}15$, $lo$, $hi$}
Barajar lista de índices $[0,\ldots,N-1]$\;
$(origen, destino) \leftarrow$ par aleatorio con $origen \neq destino$\;
$delta \leftarrow p \cdot sol[origen]$\;
$delta \leftarrow \min(delta,\; sol[origen] - lo)$\quad \textit{// no puede bajar de lo}\;
$delta \leftarrow \min(delta,\; hi - sol[destino])$\quad \textit{// no puede superar hi}\;
\Si{$delta > 0$}{
  $sol[origen] \leftarrow sol[origen] - delta$\;
  $sol[destino] \leftarrow sol[destino] + delta$\;
  evaluar fitness\;
}
\end{algorithm}

\subsection{Algoritmo Genético Generacional (AGG)}

\subsubsection*{Parámetros}
$N_p=100$, $p_c=0{,}8$, $p_m=0{,}1$ por individuo, criterio de parada: $10\,000$ evaluaciones.

\subsubsection*{Esquema de evolución y reemplazamiento}

El AGG usa la \textbf{esperanza matemática} tanto para el cruce como para la mutación:
se calculan exactamente $n_{cruces} = \lfloor p_c \cdot N_p / 2 \rfloor = 40$ pares que
se cruzan, y los $N_p/2 - 40 = 10$ pares restantes producen copias directas. La mutación
aplica exactamente $n_{mut} = \lfloor p_m \cdot N_p \rfloor = 10$ intercambios a
individuos elegidos aleatoriamente.

\begin{algorithm}[H]
\caption{AGG (generacional con elitismo, esperanza matemática)}
\KwIn{problema, maxEvals} \KwOut{mejor solución}
Inicializar $P$ con $N_p$ individuos aleatorios; $evals \leftarrow N_p$\;
\Mientras{$evals < \text{maxEvals}$}{
  Seleccionar $N_p$ padres mediante $N_p$ llamadas a \torneo{$P$}\;
  $n_{cruces} \leftarrow \lfloor 0{,}8 \cdot N_p / 2 \rfloor = 40$\;
  \Para{$i \leftarrow 1$ \KwTo $N_p/2$}{
    \Si{$i \leq n_{cruces}$}{$(h_1,h_2)\leftarrow$\cruzar{$p_{2i-1},p_{2i}$}; $evals \mathrel{+}= 2$}
    \SiNo{$h_1\leftarrow p_{2i-1}$; $h_2\leftarrow p_{2i}$}
    $P'\leftarrow P'\cup\{h_1,h_2\}$\;
  }
  Aplicar exactamente $n_{mut}=10$ mutaciones de intercambio a individuos aleatorios de $P'$\;
  \Si{$\max(P').\text{fitness} < \max(P).\text{fitness}$}{$\min(P')\leftarrow\max(P)$}
  $P \leftarrow P'$\;
}
\KwReturn{$\max(P)$}
\end{algorithm}

Se implementan dos variantes: \textbf{AGG-Arit} y \textbf{AGG-BLX}.

\subsection{Algoritmo Genético Estacionario (AGE)}

\subsubsection*{Parámetros}
$N_p=100$, $p_c=1{,}0$ (siempre se cruzan), $p_m=0{,}1$ por individuo,
criterio de parada: $10\,000$ evaluaciones.

\subsubsection*{Esquema de evolución y reemplazamiento}

\begin{algorithm}[H]
\caption{AGE (estacionario)}
\KwIn{problema, maxEvals} \KwOut{mejor solución}
Inicializar $P$ con $N_p$ individuos aleatorios; $evals \leftarrow N_p$\;
\Mientras{$evals < \text{maxEvals}$}{
  $p_1\leftarrow$\torneo{$P$}; $p_2\leftarrow$\torneo{$P$}\;
  $(h_1,h_2)\leftarrow$\cruzar{$p_1,p_2$}; $evals \mathrel{+}= 2$\;
  \mutar{$h_1$}; \mutar{$h_2$}\;
  \Si{$h_1.\text{fitness}>\min(P).\text{fitness}$}{$\min(P)\leftarrow h_1$}
  \Si{$h_2.\text{fitness}>\min(P).\text{fitness}$}{$\min(P)\leftarrow h_2$}
}
\KwReturn{$\max(P)$}
\end{algorithm}

Se implementan dos variantes: \textbf{AGE-Arit} y \textbf{AGE-BLX}.

\subsection{Algoritmos Meméticos (AM)}

Los AMs hibridan el AGG que mejor resultado ha proporcionado (\textbf{AGG-BLX},
véase Sección~\ref{sec:experimentos}) con la BL de la Práctica~1 adaptada:
fracción de movimiento $p=0{,}15$ (en lugar de $0{,}40$), máximo $100$ evaluaciones
por llamada. La BL se aplica \textbf{cada 10 generaciones}; el presupuesto de cada llamada
es $\min(100,\,\text{maxEvals}-evals)$ para respetar el límite global de $10\,000$
evaluaciones (incluidas las de la BL).

\subsubsection*{Parámetros comunes}
$N_p=100$, $p_c=0{,}8$, $p_m=0{,}1/N$ por gen.

\subsubsection*{AM-All: BL a todos los individuos}
\begin{algorithm}[H]
\caption{AM-All}
\Si{$generaciones \bmod 10 = 0$}{
  \Para{$i\leftarrow 1$ \KwTo $|P|$ \textbf{y} $evals < \text{maxEvals}$}{
    $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
    $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
  }
}
\end{algorithm}

\subsubsection*{AM-Rand: BL aleatoria al 10\% de la población}
\begin{algorithm}[H]
\caption{AM-Rand}
\Si{$generaciones \bmod 10 = 0$}{
  \Para{$i\leftarrow 1$ \KwTo $|P|$ \textbf{y} $evals < \text{maxEvals}$}{
    \Si{rand() $\leq 0{,}1$}{
      $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
      $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
    }
  }
}
\end{algorithm}

\subsubsection*{AM-Best: BL a los $0{,}1\cdot N_p$ mejores}
\begin{algorithm}[H]
\caption{AM-Best}
\Si{$generaciones \bmod 10 = 0$}{
  ordenar $P$ descendente por fitness\;
  \Para{$i\leftarrow 1$ \KwTo $\min(0{,}1\cdot|P|,|P|)$ \textbf{y} $evals < \text{maxEvals}$}{
    $budget\leftarrow\min(100,\,\text{maxEvals}-evals)$\;
    $P[i]\leftarrow$ \BL{problem, $P[i]$, budget}; $evals \mathrel{+}= \text{bl.evals}$\;
  }
}
\end{algorithm}

\section{Análisis de Resultados}

\subsection{BLX vs Aritmético}

BLX-$\alpha$ supera al cruce Aritmético en las \textbf{6 comparaciones} (3 datasets
$\times$ 2 esquemas AGG y AGE), sin excepción.

\begin{center}
\begin{tabular}{lcccc}
\toprule
\textbf{Dataset} & \textbf{AGG-Arit} & \textbf{AGG-BLX} & \textbf{AGE-Arit} & \textbf{AGE-BLX} \\
\midrule
ibex\_35 & $-4.644$ & $-4.525\;(+0.119)$ & $-4.614$ & $-4.520\;(+0.094)$ \\
syp\_100 & $-3.112$ & $-3.035\;(+0.077)$ & $-3.158$ & $-3.049\;(+0.109)$ \\
syp\_500 & $-2.961$ & $-2.877\;(+0.084)$ & $-3.201$ & $-3.095\;(+0.106)$ \\
\bottomrule
\end{tabular}
\end{center}

El cruce Aritmético es puramente explotativo: genera hijos dentro del \textit{convex hull}
de los padres. BLX puede extenderse más allá de ese convex hull (controlado por $\alpha$),
permitiendo explorar nuevas regiones del espacio de soluciones. Esta capacidad de
exploración resulta crítica tanto en espacios pequeños (escape de óptimos locales) como
en espacios de alta dimensión (diversidad necesaria). Estos resultados justifican que
los AMs usen \textbf{AGG-BLX} como base.

\subsection{AGG vs AGE: influencia del tamaño del problema}

\begin{center}
\begin{tabular}{lcc|c}
\toprule
\textbf{Dataset} & \textbf{AGG-BLX} & \textbf{AGE-BLX} & \textbf{Ganador} \\
\midrule
ibex\_35  & $-4.525$ & $-4.520$ & AGE ($+0.005$) \\
syp\_100  & $-3.035$ & $-3.049$ & AGG ($+0.014$) \\
syp\_500  & $-2.877$ & $-3.095$ & AGG ($+0.218$) \\
\bottomrule
\end{tabular}
\end{center}

El resultado más relevante es el de syp\_500: \textbf{AGG supera a AGE por un margen de
$0.218$}, mientras que en los datasets pequeños las diferencias son mínimas.

La explicación reside en la dimensionalidad: en syp\_500 ($N=457$), cada generación del
AGG genera 100 nuevos individuos que exploran el espacio de búsqueda de forma diversa,
cubriendo una mayor región del espacio en cada iteración. El AGE, que solo produce 2
hijos por iteración, tiene un ritmo de exploración muy inferior en espacios de alta
dimensión. Para ibex\_35 ($N=30$), el elitismo implícito del AGE (98 individuos intactos
por iteración) es más valioso que la exploración masiva del AGG con presupuesto limitado.

\subsection{Algoritmos Meméticos: resultados y análisis}

\textbf{AM-Best y AM-Rand son los mejores algoritmos en todos los datasets}, superando
incluso a la Búsqueda Local:

\begin{center}
\begin{tabular}{lrrr}
\toprule
\textbf{Dataset} & \textbf{BL} & \textbf{AM-Rand} & \textbf{AM-Best} \\
\midrule
ibex\_35 & $-4.573$ & $-4.491$ ($+0.082$) & $-4.491$ ($+0.082$) \\
syp\_100 & $-3.032$ & $-2.941$ ($+0.091$) & $-2.912$ ($+0.120$) \\
syp\_500 & $-3.318$ & $-2.846$ ($+0.472$) & $-2.811$ ($+0.507$) \\
\bottomrule
\end{tabular}
\end{center}

La ventaja crece con $N$: en syp\_500, AM-Best supera a BL en $0.507$ unidades de
fitness. El enfoque híbrido combina la exploración global de los AGs con la explotación
local de la BL aplicada a los mejores individuos.

\textbf{AM-All es el peor AM en todos los datasets.} La causa es el agotamiento
prematuro del presupuesto: AM-All aplica BL a los 100 individuos cada 10 generaciones,
lo que supone $100 \times 100 = 10\,000$ evaluaciones en la primera aplicación (en la
generación 10), agotando prácticamente todo el presupuesto. Apenas quedan evaluaciones
para la evolución posterior. Los individuos, todavía inmaduros en la generación 10,
quedan atrapados en óptimos locales por la BL prematura.

\subsection{Análisis de la Búsqueda Local}

Con los nuevos datos, la BL ya no es dominante. El cambio más notable es en syp\_500:
la BL alcanza $-3.318$ de media usando casi todo el presupuesto ($\approx 9840$ evals),
mientras que AGG-BLX logra $-2.877$ y AM-Best $-2.811$.

La razón es la \textbf{explosión del vecindario}: en syp\_500 ($N=457$), el número de
pares posibles para el operador de intercambio es $457 \times 456 = 208\,392$. Con un
límite de $\approx 9840$ evaluaciones, la BL solo puede explorar menos del $5\%$ de una
única pasada completa por el vecindario. En estas condiciones, la BL queda atrapada en
el primer óptimo local que encuentra, sin capacidad de exploración global.

Los AGs y AMs, en cambio, mantienen una población diversa que explora múltiples regiones
del espacio simultáneamente, siendo mucho más efectivos para $N$ grande.

\subsection{Análisis de la variabilidad ($\sigma$)}

Las desviaciones típicas de las tablas de resultados revelan patrones relevantes:

\begin{itemize}
  \item \textbf{AM-Rand y AM-Best} presentan $\sigma \approx 0.001$ en ibex\_35, lo que
  indica un comportamiento casi determinístico: todas las semillas convergen al mismo
  óptimo. La pequeña dimensionalidad ($N=30$) combinada con la BL selectiva produce una
  convergencia muy robusta.

  \item \textbf{BL} tiene $\sigma = 0.164$ en syp\_500, la mayor variabilidad entre todos
  los algoritmos en ese dataset. Esto confirma que en alta dimensión, la BL es muy
  sensible a la solución inicial: diferentes semillas llevan a óptimos locales muy
  dispares, sin que el operador pueda escapar de ellos.

  \item \textbf{Random} tiene $\sigma = 0.193$ en Fit.2025 para syp\_100, la mayor
  variabilidad global. Refleja la aleatoriedad pura sin ningún mecanismo de mejora.

  \item \textbf{Greedy} tiene $\sigma = 0.000$ en todos los datasets (determinístico).
\end{itemize}

\subsection{Análisis de los boxplots}

Los boxplots (Figuras~\ref{fig:box_ibex}--\ref{fig:box_syp500}) confirman visualmente
los resultados numéricos:

\begin{itemize}
  \item \textbf{Greedy}: línea única sin varianza (determinístico). En ibex\_35 ofrece
  buen valor medio, pero en syp\_100 y syp\_500 es el peor algoritmo, reflejando que
  la solución voraz colapsa cuando $N$ crece.

  \item \textbf{Random}: cajas anchas con bigotes largos y outliers frecuentes,
  confirmando la alta varianza. En ibex\_35 aparecen outliers superiores (semillas que
  por azar aciertan buenas combinaciones).

  \item \textbf{BL}: caja compacta en ibex\_35 y syp\_100, pero caja notablemente más
  ancha en syp\_500, reflejando la alta sensibilidad a la inicialización en alta
  dimensión ($\sigma = 0.164$).

  \item \textbf{AGs}: varianza baja y homogénea entre variantes. BLX tiene medianas
  superiores a Aritmético en todos los datasets. AGG supera a AGE en syp\_500 con una
  diferencia claramente visible en los boxplots.

  \item \textbf{AM-All}: varianza visiblemente mayor que AM-Rand y AM-Best
  (especialmente en syp\_100 y syp\_500), reflejando que el agotamiento prematuro del
  presupuesto hace que el resultado dependa más de la semilla inicial.

  \item \textbf{AM-Rand y AM-Best}: distribuciones con medianas superiores a todos los
  AGs puros y a la BL, con varianza reducida. AM-Best tiene medianas ligeramente
  superiores a AM-Rand, confirmando la eficiencia de aplicar BL a los mejores individuos.
\end{itemize}

\subsection{Capacidad de generalización (test 2025)}

Los rankings de entrenamiento se mantienen en test para los algoritmos más robustos.
AM-Best generaliza bien en todos los datasets: ibex\_35 test $-3.084$ (mejor de todos),
syp\_100 test $-3.725$ (mejor), syp\_500 test $-3.907$ (mejor). AGG-BLX y AGE-BLX
también generalizan correctamente, con test próximos a los de entrenamiento.

El caso más llamativo es \textbf{Greedy en syp\_100 y syp\_500}: el fitness de test
($-5.020$ y $-5.205$) es muy inferior al de entrenamiento ($-3.664$ y $-3.438$),

\subsection{Conclusiones generales}

\begin{enumerate}
  \item \textbf{BLX supera a Aritmético} en las 6 comparaciones (3 datasets $\times$
  2 esquemas): la exploración fuera del convex hull es crítica para la calidad.

  \item \textbf{AGE supera a AGG} para $N$ pequeño (ibex\_35); \textbf{AGG supera a AGE}
  para $N$ grande (syp\_500, margen $0.218$). La exploración masiva del AGG es decisiva
  en alta dimensión.

  \item \textbf{AM-Best y AM-Rand son los mejores algoritmos globales}, superando a BL
  en todos los datasets. La hibridación evolutivo-local es la estrategia ganadora.

  \item \textbf{AM-All fracasa} por agotamiento del presupuesto en la generación 10:
  $100 \times 100 = 10\,000$ evaluaciones de BL consumen todo el presupuesto sin
  evolución suficiente previa.

  \item \textbf{BL no es competitiva para $N$ grande}: la explosión del vecindario
  ($457 \times 456 = 208\,392$ pares en syp\_500) impide explorar más del $5\%$ del
  vecindario con el presupuesto disponible.
\end{enumerate}

\subsection{Análisis}

\textbf{$\alpha = 0.5$ es óptimo en los tres datasets}, con la mejor fitness de
entrenamiento en todos los casos. La curva de calidad tiene forma de \textbf{U invertida}:
mejora monótonamente desde $\alpha = 0.0$ hasta $\alpha \approx 0.5$, y luego se degrada
a $\alpha = 1.0$. Este comportamiento es consistente en los tres problemas.

Un hallazgo contraintuitivo es que \textbf{$\sigma$ disminuye al aumentar $\alpha$} hasta
el óptimo: más exploración conduce a resultados más consistentes, no más variables. La
explicación es que $\alpha$ bajo produce convergencia prematura a diferentes óptimos
locales (alta varianza), mientras que $\alpha$ adecuado permite escapar de dichos óptimos
y converger hacia el mismo atractor global.

El caso más extremo es ibex\_35 con $\alpha \in \{0.5, 0.7\}$: $\sigma = 0.000$,
prácticamente determinístico en las 50 semillas. Esto indica \textbf{convergencia
prematura} a un único atractor: el espacio de soluciones pequeño ($N=30$, límites
estrechos) combinado con alta exploración hace que todos los runs converjan al mismo
punto. Si bien el valor encontrado es el mejor, la ausencia de diversidad puede ser
problemática en paisajes de fitness más complejos.

El valor $\alpha = 0.3$ recomendado en el guion es subóptimo: $\alpha = 0.5$ mejora el
fitness en aproximadamente un $7\%$ tanto en syp\_100 ($-3.035$ vs $-2.834$) como en
syp\_500 ($-2.877$ vs $-2.682$).

Finalmente, $\alpha = 0.0$ se comporta como el cruce Aritmético puro (sin extensión del
intervalo), confirmando que la ventaja de BLX frente a Aritmético proviene exclusivamente
de la capacidad de exploración más allá del convex hull de los padres.
